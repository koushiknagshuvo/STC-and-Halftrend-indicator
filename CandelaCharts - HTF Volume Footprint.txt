method text_size(string size) =>
    switch size
        "Tiny"   => size.tiny
        "Small"  => size.small
        "Normal" => size.normal
        "Large"  => size.large
        "Huge"   => size.huge
        "Auto"   => size.auto

method font_style(string font) =>
    switch font
        "Default"   => font.family_default
        "Monospace" => font.family_monospace

minute_label(int mins) =>
    switch
        mins >= 60 and mins % 60 == 0 => str.tostring(int(mins / 60)) + "H"
        => str.tostring(mins) + "m"

tf_format(string tf) =>
    string res = tf
    if tf == "D" or tf == "1D"
        res := "1D"
    else if tf == "W" or tf == "1W"
        res := "1W"
    else if tf == "M" or tf == "1M"
        res := "1M"
    else
        float mins = str.tonumber(tf)
        if not na(mins)
            res := minute_label(int(mins))
    res

is_dark() =>
    chart.bg_color == color.black or chart.bg_color == #131722

format_volume(float vol, bool compact) =>
    if not compact
        str.tostring(math.round(vol))
    else if vol >= 1000000
        str.tostring(vol / 1000000, "#.#") + "M"
    else if vol >= 1000
        str.tostring(vol / 1000, "#.#") + "K"
    else
        str.tostring(math.round(vol))

pad_volume(string txt) =>
    int len = str.length(txt)
    switch
        len == 1 => "   "
        len == 2 => "  "
        len == 3 => " "
        => ""

format_bid_ask(float bid_v, float ask_v, bool compact) =>
    string s_bid = format_volume(bid_v, compact)
    string s_ask = format_volume(ask_v, compact)
    pad_volume(s_bid) + s_bid + " " + s_ask + pad_volume(s_ask)

calc_volume_ratio(float vol, float ref_max, string mode) =>
    float ratio = 0.0
    if vol > 0.0 and ref_max > 0.0
        float raw = math.min(1.0, vol / ref_max)
        ratio := switch mode
            "Square Root" => math.sqrt(raw)
            "Logarithmic" => math.log(1.0 + 9.0 * raw) / math.log(10.0)
            "Linear"      => raw
            => math.sqrt(raw)
    ratio

bar_length(float ratio, int half_bars) =>
    int(math.max(1, math.round(ratio * half_bars)))

bar_colors(UDT_Palette pal, bool is_bid, float ratio) =>
    color bg = is_bid ? pal.bid_solid_bg : pal.ask_solid_bg
    color bd = is_bid ? pal.bid_solid_bd : pal.ask_solid_bd

    if hvf_fp_gradient
        bg := color.from_gradient(ratio, 0.0, 1.0, is_bid ? pal.bid_bg_lo : pal.ask_bg_lo, is_bid ? pal.bid_bg_hi : pal.ask_bg_hi)
        bd := color.from_gradient(ratio, 0.0, 1.0, is_bid ? pal.bid_bd_lo : pal.ask_bd_lo, is_bid ? pal.bid_bd_hi : pal.ask_bd_hi)

    [bg, bd]

method update_candle(HTF_Footprint_Candle candle) =>
    candle.c := close
    candle.ct := time
    candle.c_idx := bar_index
    candle.h := math.max(candle.h, high)
    candle.l := math.min(candle.l, low)
    candle.is_bull := candle.c >= candle.o

method settle_candle(HTF_Footprint_Candle candle) =>
    candle.is_closed := true
    candle.ct := time[1]
    candle.c_idx := bar_index - 1

method accumulate_ltf(HTF_Footprint_Candle candle, UDT_LTF_Tape tape) =>
    if not na(tape.c) and tape.c.size() > 0
        for i = 0 to tape.c.size() - 1
            candle.ltf_c.push(tape.c.get(i))
            candle.ltf_o.push(tape.o.get(i))
            candle.ltf_h.push(tape.h.get(i))
            candle.ltf_l.push(tape.l.get(i))
            candle.ltf_v.push(tape.v.get(i))
    else
        candle.ltf_c.push(close)
        candle.ltf_o.push(open)
        candle.ltf_h.push(high)
        candle.ltf_l.push(low)
        candle.ltf_v.push(volume)

// # |   Order Flow Imbalances   | #
// # ========================================================================= #

//#region

method compute_imbalances(HTF_Footprint_Candle candle, int num_rows, float imb_ratio, float min_imb_vol) =>
    int buy_cnt = 0
    int sell_cnt = 0
    if num_rows > 1
        for r = 0 to num_rows - 2
            float curr_bid = candle.bid_vols.get(r)
            float next_ask = candle.ask_vols.get(r + 1)

            // Buying Imbalance: aggressive ask volume outweighs diagonal lower bid
            if next_ask >= curr_bid * imb_ratio and next_ask >= min_imb_vol and next_ask > 0.0
                candle.buy_imbalances.set(r + 1, true)
                buy_cnt += 1

            // Selling Imbalance: aggressive bid volume outweighs diagonal higher ask
            if curr_bid >= next_ask * imb_ratio and curr_bid >= min_imb_vol and curr_bid > 0.0
                candle.sell_imbalances.set(r, true)
                sell_cnt += 1

    candle.buy_imb_count := buy_cnt
    candle.sell_imb_count := sell_cnt

method reset_profile(HTF_Footprint_Candle candle) =>
    candle.bin_lows.clear()
    candle.bin_highs.clear()
    candle.bid_vols.clear()
    candle.ask_vols.clear()
    candle.tot_vols.clear()
    candle.buy_imbalances.clear()
    candle.sell_imbalances.clear()
    candle.buy_imb_count := 0
    candle.sell_imb_count := 0
    candle.poc_bin := 0
    candle.poc_price := candle.c
    candle.max_side_vol := 0.0
    candle.avg_bin_vol := 0.0
    candle.va_high_bin := 0
    candle.va_low_bin := 0
    candle.vah_price := candle.h
    candle.val_price := candle.l

method seed_bins(HTF_Footprint_Candle candle, int num_rows, float price_low, float step) =>
    for r = 0 to num_rows - 1
        float bin_low = price_low + r * step
        candle.bin_lows.push(bin_low)
        candle.bin_highs.push(bin_low + step)
        candle.bid_vols.push(0.0)
        candle.ask_vols.push(0.0)
        candle.tot_vols.push(0.0)
        candle.buy_imbalances.push(false)
        candle.sell_imbalances.push(false)

method add_to_bin(HTF_Footprint_Candle candle, int idx, float buy_v, float sell_v) =>
    candle.bid_vols.set(idx, candle.bid_vols.get(idx) + sell_v)
    candle.ask_vols.set(idx, candle.ask_vols.get(idx) + buy_v)

// Distributes a single lower timeframe sample across the profile bins by price overlap
method add_slice(HTF_Footprint_Candle candle, int num_rows, float price_low, float step, int idx) =>
    float s_close = candle.ltf_c.get(idx)
    float s_open = candle.ltf_o.get(idx)
    float s_high = candle.ltf_h.get(idx)
    float s_low = candle.ltf_l.get(idx)
    float s_vol = candle.ltf_v.get(idx)

    bool is_buy = s_close > s_open
    bool is_sell = s_close < s_open
    float buy_v = is_buy ? s_vol : is_sell ? 0.0 : s_vol * 0.5
    float sell_v = is_sell ? s_vol : is_buy ? 0.0 : s_vol * 0.5

    float s_range = s_high - s_low
    if s_range <= 0.0
        int bin_idx = int(math.floor((s_close - price_low) / step))
        candle.add_to_bin(math.max(0, math.min(num_rows - 1, bin_idx)), buy_v, sell_v)
    else
        for r = 0 to num_rows - 1
            float overlap = math.min(s_high, candle.bin_highs.get(r)) - math.max(s_low, candle.bin_lows.get(r))
            if overlap > 0.0
                float frac = overlap / s_range
                candle.add_to_bin(r, buy_v * frac, sell_v * frac)

// Resolves POC, max side volume and average bin volume; returns total traded volume
method compute_stats(HTF_Footprint_Candle candle, int num_rows, float step) =>
    float max_bin = 0.0
    float max_side = 0.0
    float sum_v = 0.0
    int poc_bin = 0

    for r = 0 to num_rows - 1
        float bid_v = candle.bid_vols.get(r)
        float ask_v = candle.ask_vols.get(r)
        float total = bid_v + ask_v
        candle.tot_vols.set(r, total)
        sum_v += total
        if total > max_bin
            max_bin := total
            poc_bin := r
        max_side := math.max(max_side, math.max(bid_v, ask_v))

    candle.poc_bin := poc_bin
    candle.poc_price := candle.bin_lows.get(poc_bin) + step * 0.5
    candle.max_side_vol := max_side > 0.0 ? max_side : max_bin
    candle.avg_bin_vol := num_rows > 0 ? sum_v / num_rows : 0.0
    sum_v

// Auction Market Theory value area: expands 70% of volume outward from the POC
method compute_value_area(HTF_Footprint_Candle candle, int num_rows, float sum_v) =>
    int va_hi = candle.poc_bin
    int va_lo = candle.poc_bin

    if sum_v > 0.0 and num_rows > 0
        float target_vol = sum_v * 0.70
        float acc_vol = candle.tot_vols.get(candle.poc_bin)

        while acc_vol < target_vol and (va_hi < num_rows - 1 or va_lo > 0)
            float up_vol = va_hi < num_rows - 1 ? candle.tot_vols.get(va_hi + 1) : 0.0
            float dn_vol = va_lo > 0 ? candle.tot_vols.get(va_lo - 1) : 0.0

            if va_hi < num_rows - 1 and (va_lo == 0 or up_vol >= dn_vol)
                va_hi += 1
                acc_vol += up_vol
            else if va_lo > 0
                va_lo -= 1
                acc_vol += dn_vol
            else
                break

    candle.va_high_bin := va_hi
    candle.va_low_bin := va_lo
    candle.vah_price := candle.bin_highs.size() > va_hi ? candle.bin_highs.get(va_hi) : candle.h
    candle.val_price := candle.bin_lows.size() > va_lo ? candle.bin_lows.get(va_lo) : candle.l

method compute_footprint(HTF_Footprint_Candle candle, int num_rows) =>
    candle.reset_profile()

    float step = (candle.h - candle.l) / math.max(1, num_rows)
    if step > 0 and candle.ltf_v.size() > 0
        candle.seed_bins(num_rows, candle.l, step)
        for i = 0 to candle.ltf_v.size() - 1
            candle.add_slice(num_rows, candle.l, step, i)
        float sum_v = candle.compute_stats(num_rows, step)
        candle.compute_value_area(num_rows, sum_v)
        candle.compute_imbalances(num_rows, hvf_imbalance_ratio, hvf_imbalance_min_vol)

method show_row_numbers(HTF_Footprint_Candle candle, int r, float tot_v) =>
    bool show = false
    switch hvf_fp_num_mode
        "All Rows"                 => show := true
        "Significant Only (> Avg)" => show := tot_v >= candle.avg_bin_vol
        "POC Only"                 => show := r == candle.poc_bin
        "Imbalances Only"          => show := candle.buy_imbalances.get(r) or candle.sell_imbalances.get(r)
    show

method draw_row(HTF_Footprint_Candle candle, int r, UDT_Store s, UDT_Palette pal, int fp_center, int half_fp_bars, float ref_max) =>
    float bin_low = candle.bin_lows.get(r)
    float bin_high = candle.bin_highs.get(r)
    float bid_v = candle.bid_vols.get(r)
    float ask_v = candle.ask_vols.get(r)
    float tot_v = bid_v + ask_v

    float ratio_b = calc_volume_ratio(bid_v, ref_max, hvf_fp_scale_mode)
    float ratio_a = calc_volume_ratio(ask_v, ref_max, hvf_fp_scale_mode)
    int bid_len = bid_v > 0.0 ? bar_length(ratio_b, half_fp_bars) : 0
    int ask_len = ask_v > 0.0 ? bar_length(ratio_a, half_fp_bars) : 0

    // Bid volume bar (extends left from the center split)
    if bid_len > 0
        [b_bg, b_bd] = bar_colors(pal, true, ratio_b)
        s.bin_boxes.push(make_box(fp_center - bid_len, bin_high, fp_center, bin_low, b_bg, b_bd, 1))

    // Ask volume bar (extends right from the center split)
    if ask_len > 0
        [a_bg, a_bd] = bar_colors(pal, false, ratio_a)
        s.bin_boxes.push(make_box(fp_center, bin_high, fp_center + ask_len, bin_low, a_bg, a_bd, 1))

    // Imbalance accent highlight
    if hvf_imbalance_highlight
        if candle.buy_imbalances.get(r)
            int ask_imb_len = math.max(1, ask_len)
            s.bin_boxes.push(make_box(fp_center, bin_high, fp_center + ask_imb_len, bin_low, color.new(theme_bull_color, 75), theme_bull_color, 2))
        if candle.sell_imbalances.get(r)
            int bid_imb_len = math.max(1, bid_len)
            s.bin_boxes.push(make_box(fp_center - bid_imb_len, bin_high, fp_center, bin_low, color.new(theme_bear_color, 75), theme_bear_color, 2))

    // High Volume Node (POC) accent border
    if hvf_fp_highlight_poc and r == candle.poc_bin
        s.bin_boxes.push(make_box(
           fp_center - math.max(1, bid_len),
           bin_high,
           fp_center + math.max(1, ask_len),
           bin_low,
           color.new(hvf_poc_color, 75),
           hvf_poc_color,
           2
           ))

    // Monospace Bid x Ask overlay
    if hvf_fp_show_numbers and (bid_v > 0 or ask_v > 0) and candle.show_row_numbers(r, tot_v)
        color lbl_css = r == candle.poc_bin ? hvf_poc_color : (candle.buy_imbalances.get(r) ? theme_bull_color : (candle.sell_imbalances.get(r) ? theme_bear_color : hvf_num_color))
        s.bin_labels.push(make_label(
           fp_center,
           (bin_low + bin_high) * 0.5,
           format_bid_ask(bid_v, ask_v, hvf_fp_compact_num),
           lbl_css,
           label.style_label_center,
           general_text
           ))

method draw_footprint(HTF_Footprint_Candle candle, UDT_Store s, UDT_Palette pal, int fp_center, int half_fp_bars) =>
    int bins = candle.bin_lows.size()
    if bins > 0
        float ref_max = math.max(1.0, candle.max_side_vol)
        for r = 0 to bins - 1
            candle.draw_row(r, s, pal, fp_center, half_fp_bars, ref_max)

method draw_value_area(HTF_Footprint_Candle candle, UDT_Store s, int fp_center, int half_fp_bars) =>
    if not na(candle.vah_price) and not na(candle.val_price)
        int fp_left = fp_center - half_fp_bars
        int fp_right = fp_center + half_fp_bars

        s.draw_price_line(fp_left, fp_right, candle.vah_price, hvf_va_color, 2)
        s.draw_price_line(fp_left, fp_right, candle.val_price, hvf_va_color, 2)

        if hvf_show_va_labels
            s.draw_level_label(fp_right, candle.vah_price, "VAH", "Value Area High", hvf_va_color)
            s.draw_level_label(fp_right, candle.val_price, "VAL", "Value Area Low", hvf_va_color)
            if not na(candle.poc_price)
                s.draw_level_label(fp_right, candle.poc_price, "POC", "Point of Control", hvf_poc_color)

method draw_body(HTF_Footprint_Candle candle, UDT_Store s, int c_left, int c_right) =>
    color css = candle.is_bull ? theme_bull_color : theme_bear_color
    float body_top = math.max(candle.o, candle.c)
    float body_bot = math.min(candle.o, candle.c)
    int wick_x = math.round(math.avg(c_left, c_right))

    s.bin_boxes.push(make_box(c_left, body_top, c_right, body_bot, css, css, 1))
    if candle.h > body_top
        s.bin_lines.push(make_line(wick_x, body_top, wick_x, candle.h, css, 1))
    if body_bot > candle.l
        s.bin_lines.push(make_line(wick_x, body_bot, wick_x, candle.l, css, 1))

render_htf_group(UDT_Group group, int base_offset, int rows, UDT_Layout layout, UDT_Palette pal, UDT_Store s) =>
    int total = math.min(group.candles.size(), group.count)

    if total > 0
        float group_max_h = -1.0e10
        int min_x = 999999999
        int max_x = -999999999

        for i = 0 to total - 1
            HTF_Footprint_Candle candle = group.candles.get(i)
            candle.compute_footprint(rows)

            int shift = base_offset + (total - 1 - i) * layout.unit_width
            int candle_center = last_bar_index + shift + layout.candle_half_w
            int c_left = candle_center - layout.candle_half_w
            int c_right = candle_center + layout.candle_half_w
            int fp_center = c_right + hvf_candle_fp_space + 1 + layout.half_fp_bars

            group_max_h := math.max(group_max_h, candle.h)
            min_x := math.min(min_x, c_left)
            max_x := math.max(max_x, fp_center + layout.half_fp_bars)

            candle.draw_body(s, c_left, c_right)

            // Footprint center split divider
            if candle.bin_highs.size() > 0
                s.bin_lines.push(make_line(
                   fp_center,
                   candle.bin_highs.last(),
                   fp_center,
                   candle.bin_lows.first(),
                   color.new(hvf_wick_color, 60),
                   1,
                   line.style_dotted
                   ))

            candle.draw_footprint(s, pal, fp_center, layout.half_fp_bars)
            candle.draw_value_area(s, fp_center, layout.half_fp_bars)

        // Centered timeframe header tag above the candle + ladder cluster
        if hvf_show_labels and group_max_h > -1.0e9
            int group_center = math.round((min_x + max_x) * 0.5)
            s.bin_labels.push(make_label(group_center, group_max_h, tf_format(group.tf), chart.fg_color, label.style_label_down, "Large"))

draw_mapped_group(UDT_Group group, color bull_c, color bear_c, bool is_group2, UDT_Store s) =>
    int total = math.min(group.candles.size(), group.count)

    if total > 0
        for i = 0 to total - 1
            HTF_Footprint_Candle candle = group.candles.get(i)
            if na(candle.o) or na(candle.h) or na(candle.l) or na(candle.c)
                continue

            int x1 = candle.o_idx
            int x2 = candle.is_closed ? candle.c_idx : bar_index
            if x2 < x1
                x2 := x1

            color css = candle.is_bull ? bull_c : bear_c
            float body_top = math.max(candle.o, candle.c)
            float body_bot = math.min(candle.o, candle.c)
            int wick_x = math.round(math.avg(x1, x2))

            s.bin_boxes.push(make_box(x1, body_top, x2, body_bot, color.new(css, 85), css, 1))

            if candle.h > body_top
                s.bin_lines.push(make_line(wick_x, body_top, wick_x, candle.h, css, 1))
            if candle.l < body_bot
                s.bin_lines.push(make_line(wick_x, body_bot, wick_x, candle.l, css, 1))

            if hvf_map_trend
                s.bin_lines.push(make_line(x1, candle.o, x2, candle.c, css, 1))

            if hvf_map_sep and x1 > 0
                s.bin_lines.push(make_line(x1, candle.h, x1, candle.l, color.new(hvf_wick_color, 70), 1, line.style_dotted, true))

            if hvf_show_labels
                string tf_txt = is_group2 ? tf_format(group.tf) + "\n \n " : tf_format(group.tf) + "\n"
                s.bin_labels.push(make_label(wick_x, candle.h, tf_txt, css, label.style_label_down, general_text))

advance_group(UDT_Group group, bool is_new_period, UDT_LTF_Tape tape) =>
    array<HTF_Footprint_Candle> candles = group.candles

    if is_new_period or candles.size() == 0
        if candles.size() > 0
            candles.first().settle_candle()
        if candles.size() >= group.count
            candles.pop()
        candles.unshift(new_candle(group.tf))
    else if candles.size() > 0
        candles.first().update_candle()

    if candles.size() > 0
        candles.first().accumulate_ltf(tape)
