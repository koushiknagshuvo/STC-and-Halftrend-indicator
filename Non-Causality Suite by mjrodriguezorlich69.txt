solvePentadiagonal(array<float> b, array<float> d, array<float> c, array<float> f, int n, array<float> x) =>
    var array<float> d_tmp = array.new_float(n)
    var array<float> c_tmp = array.new_float(n)
    var array<float> b_tmp = array.new_float(n)
    
    for i = 0 to n - 1
        d_tmp.set(i, d.get(i)), c_tmp.set(i, c.get(i)), b_tmp.set(i, b.get(i))

    float r = c_tmp.get(0), s = c_tmp.get(1), t = f.get(0), xmult = 0.0
    int l1 = n - 1, l2 = n - 2

    for i = 1 to l2
        float d_prev = d_tmp.get(i - 1)
        xmult := r / d_prev
        d_tmp.set(i, d_tmp.get(i) - xmult * c_tmp.get(i - 1))
        c_tmp.set(i, c_tmp.get(i) - xmult * f.get(i - 1))
        b_tmp.set(i, b_tmp.get(i) - xmult * b_tmp.get(i - 1))
        xmult := t / d_prev
        r     := s - xmult * c_tmp.get(i - 1)
        d_tmp.set(i + 1, d_tmp.get(i + 1) - xmult * f.get(i - 1))
        b_tmp.set(i + 1, b_tmp.get(i + 1) - xmult * b_tmp.get(i - 1))
        if i < l2
            s := c_tmp.get(i + 1), t := f.get(i)
        
    xmult := r / d_tmp.get(l2)
    d_tmp.set(l1, d_tmp.get(l1) - xmult * c_tmp.get(l2))
    x.set(l1, (b_tmp.get(l1) - xmult * b_tmp.get(l2)) / d_tmp.get(l1))
    x.set(l2, (b_tmp.get(l2) - c_tmp.get(l2) * x.get(l1)) / d_tmp.get(l2))
    
    for i = n - 3 to 0 by 1
        x.set(i, (b_tmp.get(i) - f.get(i) * x.get(i + 2) - c_tmp.get(i) * x.get(i + 1)) / d_tmp.get(i))

whittakerALS(array<float> prices, array<float> d_base, array<float> c_base, array<float> f_base, int n, int iters, float p, array<float> z) =>
    var array<float> weights = array.new_float(lengthInput, 1.0)
    var array<float> b_curr  = array.new_float(lengthInput)
    var array<float> d_curr  = array.new_float(lengthInput)
    weights.fill(1.0)
    
    for it = 1 to iters
        for i = 0 to n - 1
            float w = weights.get(i)
            d_curr.set(i, d_base.get(i) + w)
            b_curr.set(i, w * prices.get(i))
        solvePentadiagonal(b_curr, d_curr, c_base, f_base, n, z)
        for i = 0 to n - 1
            weights.set(i, prices.get(i) > z.get(i) ? p : 1.0 - p)

nadarayaWatson(array<float> src, int n, float h, array<float> out) =>
    for i = 0 to n - 1
        float sum_w = 0.0, sum_xw = 0.0
        for j = 0 to n - 1
            float diff = i - j
            float weight = math.exp(-(diff * diff) / (2.0 * h * h))
            sum_w += weight
            sum_xw += src.get(j) * weight
        out.set(i, sum_xw / sum_w)

// Dynamic Non-Causal Savitzky-Golay Filter (Quadratic/Cubic Gram Polynomials)
sgFilter(array<float> src, int n, int m, array<float> out) =>
    for i = 0 to n - 1
        float sum_val = 0.0
        float sum_w = 0.0
        for j = -m to m
            int idx = i + j
            if idx >= 0 and idx < n
                float w = 3.0 * (3.0 * m * m + 3.0 * m - 1.0 - 5.0 * j * j)
                sum_val += src.get(idx) * w
                sum_w += w
        out.set(i, sum_val / sum_w)

calc_atan2(float y, float x) =>
    float res = 0.0
    if x > 0.0
        res := math.atan(y / x)
    else if x < 0.0 and y >= 0.0
        res := math.atan(y / x) + math.pi
    else if x < 0.0 and y < 0.0
        res := math.atan(y / x) - math.pi
    else if x == 0.0 and y > 0.0
        res := math.pi / 2.0
    else if x == 0.0 and y < 0.0
        res := -math.pi / 2.0
    res

                    int tr_i = bar_index - track_idx

                    float tr_sg_slope = sgEnable ? (zSG.get(tr_i) - (tr_i + 1 < lengthInput ? zSG.get(tr_i + 1) : zSG.get(tr_i))) : 0.0

                    bool tr_sg_bear = not sgEnable or tr_sg_slope <= 0

                    label.new(track_idx, na, text = "▼", yloc = yloc.abovebar, style = label.style_none, textcolor = color.new(BEAR_COLOR, tr_transp_bear), size = size.small, tooltip=tooltip_top, force_overlay = true)

                if val_eD >= track_eD or weso_val > track_weso

                    float tr_sg_slope = sgEnable ? (zSG.get(tr_i) - (tr_i + 1 < lengthInput ? zSG.get(tr_i + 1) : zSG.get(tr_i))) : 0.0

                    bool tr_sg_bull = not sgEnable or tr_sg_slope >= 0

                    float tr_hist_std  = arr_meta_std.get(tr_i)
                    float tr_z_score   = tr_hist_std != 0 ? (tr_val_c - tr_hist_mean) / tr_hist_std : 0.0

                    log_times.push(str.format("{0,date,MM/dd HH:mm}", time[tr_i]))

                    label.new(track_idx, na, text = "▲", yloc = yloc.belowbar, style = label.style_none, textcolor = color.new(BULL_COLOR, tr_transp_bull), size = size.small, tooltip=tooltip_bot, force_overlay = true)

        if not na(val_prev)

        val_prev := prev_ema

        if prev_ema >= 0.0
            if slope > 0.0
                hist_bull_acc.push(chart.point.from_index(idx, 0.0)), hist_bull_acc.push(chart.point.from_index(idx, prev_ema)), hist_bull_acc.push(chart.point.from_index(idx, 0.0))
            else
                hist_bull_dec.push(chart.point.from_index(idx, 0.0)), hist_bull_dec.push(chart.point.from_index(idx, prev_ema)), hist_bull_dec.push(chart.point.from_index(idx, 0.0))
        else
            if slope > 0.0
                hist_bear_dec.push(chart.point.from_index(idx, 0.0)), hist_bear_dec.push(chart.point.from_index(idx, prev_ema)), hist_bear_dec.push(chart.point.from_index(idx, 0.0))
            else
                hist_bear_acc.push(chart.point.from_index(idx, 0.0)), hist_bear_acc.push(chart.point.from_index(idx, prev_ema)), hist_bear_acc.push(chart.point.from_index(idx, 0.0))
                
        // Subpane WESO Squeeze Dots
        if i < 250
            color dot_col = spreadRatio < sqzThresholdInput ? BEAR_COLOR : BULL_COLOR
            label.new(x = idx, y = 0.0, text = "•", style = label.style_none, textcolor = dot_col, textalign = text.align_center, tooltip="WESO Squeeze Compression")

        float hist_std  = arr_meta_std.get(i)

        if hilbertEnable
            float current_nw = zNW.get(i)

            float z_score   = hist_std != 0 ? (val_c - hist_mean) / hist_std : 0.0
            string z_str    = (z_score > 0 ? "+" : "") + str.tostring(z_score, "#.##")
            
            int conf_bull_curr = transp_bull == 0 ? 100 : (transp_bull == 50 ? 50 : 15)
            int conf_bear_curr = transp_bear == 0 ? 100 : (transp_bear == 50 ? 50 : 15)
            
            if not na(track_sine) and not na(track_lead)
                if track_sine <= track_lead and sine_val > lead_val
                    log_bars.push(idx)
                    log_types.push("Phase Bull ◆")
                    log_times.push(str.format("{0,date,MM/dd HH:mm}", time[i]))
                    log_zs.push(z_str)
                    log_confs.push(str.tostring(conf_bull_curr) + "%")
                    log_cols.push(#00FFFF)

                    label.new(idx, na, text = "◆", yloc = yloc.belowbar, style = label.style_none, textcolor = color.new(#00FFFF, transp_bull), size = size.small, tooltip=tt_phase_bull, force_overlay = true)

                    log_bars.push(idx)
                    log_types.push("Phase Bear ◆")
                    log_times.push(str.format("{0,date,MM/dd HH:mm}", time[i]))
                    log_zs.push(z_str)
                    log_confs.push(str.tostring(conf_bear_curr) + "%")
                    log_cols.push(#FF00FF)

                    label.new(idx, na, text = "◆", yloc = yloc.abovebar, style = label.style_none, textcolor = color.new(#FF00FF, transp_bear), size = size.small, tooltip=tt_phase_bear, force_overlay = true)

        bool is_armed_long_global  = val_c < hist_mean - (hist_std * 1.5)

                    last_armed_cvd_short := curr_ltf_cvd

                last_armed_cvd_short := na

                        running_bot_stack += 1
                        last_bot_idx := idx
                        trig_cvd_long := true
                    last_armed_price_long := val_c
                    last_armed_cvd_long := curr_ltf_cvd

                last_armed_price_long := na
                last_armed_cvd_long := na

                        label.new(idx, y_pos, "O\nI\n▼", color=color.new(BEAR_COLOR, 0), textcolor=color.white, style=label.style_label_down, size=size.tiny, tooltip="Open Interest Exhaustion (Short)\nLiquidity Engine Failure", force_overlay = true)

                        last_top_idx := idx

                    last_armed_price_short_oi := val_c
                    last_armed_oi_short := curr_ltf_oi
            else

                last_armed_oi_short := na
                trig_oi_short := false

                if na(last_armed_price_long_oi) or val_c < last_armed_price_long_oi
                    if not na(last_armed_oi_long) and curr_ltf_oi < last_armed_oi_long and not trig_oi_long
                        float y_pos = arr_lows.get(i) - (running_bot_stack * stack_offset)
                        label.new(idx, y_pos, "O\nI\n▲", color=color.new(BULL_COLOR, 0), textcolor=color.white, style=label.style_label_up, size=size.tiny, tooltip="Open Interest Exhaustion (Long)\nLiquidity Engine Failure", force_overlay = true)
                        running_bot_stack += 1
                        last_bot_idx := idx
                        trig_oi_long := true
                    last_armed_price_long_oi := val_c
                    last_armed_oi_long := curr_ltf_oi
            else
                last_armed_price_long_oi := na
                last_armed_oi_long := na
                trig_oi_long := false

        // 4. Macro Capital Fracture (USDT.D)
        if useUsdtExec
            float curr_ltf_usdt = arr_ltf_usdt.get(i)
            
            if is_armed_short_global
                if na(last_armed_price_short_usdt) or val_c > last_armed_price_short_usdt
                    if not na(last_armed_usdt_short) and curr_ltf_usdt >= last_armed_usdt_short and not trig_usdt_short
                        float y_pos = arr_highs.get(i) + (running_top_stack * stack_offset)
                        label.new(idx, y_pos, "U\nS\nD\nT\n▼", color=color.new(BEAR_COLOR, 0), textcolor=color.white, style=label.style_label_down, size=size.tiny, tooltip="Macro Capital Fracture (USDT.D)\nArmed by Non-Causal Boundary", force_overlay = true)
                        running_top_stack += 1
                        last_top_idx := idx
                        trig_usdt_short := true
                    last_armed_price_short_usdt := val_c
                    last_armed_usdt_short := curr_ltf_usdt
            else
                last_armed_price_short_usdt := na
                last_armed_usdt_short := na
                trig_usdt_short := false
                
            if is_armed_long_global
                if na(last_armed_price_long_usdt) or val_c < last_armed_price_long_usdt
                    if not na(last_armed_usdt_long) and curr_ltf_usdt <= last_armed_usdt_long and not trig_usdt_long
                        float y_pos = arr_lows.get(i) - (running_bot_stack * stack_offset)
                        label.new(idx, y_pos, "U\nS\nD\nT\n▲", color=color.new(BULL_COLOR, 0), textcolor=color.white, style=label.style_label_up, size=size.tiny, tooltip="Macro Capital Fracture (USDT.D)\nArmed by Non-Causal Boundary", force_overlay = true)

                    last_armed_usdt_long := curr_ltf_usdt

                last_armed_usdt_long := na

                if not was_armed_short

                    cum_tp_vol_short := 0.0
                    trig_vwap_short  := false

                cum_tp_vol_short += curr_ltf_v * curr_ltf_h3

                    if prev_ltf_c >= prev_vwap_short and curr_ltf_c < curr_vwap_short and not trig_vwap_short

                        running_top_stack += 1
                        last_top_idx := idx
                        trig_vwap_short := true
                prev_vwap_short := curr_vwap_short
            was_armed_short := is_armed_short_global
            
            // Long Reclaim
            if is_armed_long_global
                if not was_armed_long
                    cum_vol_long    := 0.0
                    cum_tp_vol_long := 0.0
                    trig_vwap_long  := false
                cum_vol_long    += curr_ltf_v
                cum_tp_vol_long += curr_ltf_v * curr_ltf_h3
                float curr_vwap_long = cum_vol_long > 0 ? cum_tp_vol_long / cum_vol_long : curr_ltf_c
                
                if was_armed_long and not na(prev_ltf_c) and not na(prev_vwap_long)
                    if prev_ltf_c <= prev_vwap_long and curr_ltf_c > curr_vwap_long and not trig_vwap_long
                        float y_pos = arr_lows.get(i) - (running_bot_stack * stack_offset)

                        running_bot_stack += 1
                        last_bot_idx := idx
                        trig_vwap_long := true
                prev_vwap_long := curr_vwap_long
            was_armed_long := is_armed_long_global
            
            prev_ltf_c := curr_ltf_c

    // 4. Render Final Polylines (Subpane)
    polyline.new(weso_pts, line_color = color.blue, line_width = 2)
    polyline.new(hist_bull_acc, line_color = COL_BULL_ACC, line_width = 3)
    polyline.new(hist_bull_dec, line_color = COL_BULL_DEC, line_width = 3)
    polyline.new(hist_bear_acc, line_color = COL_BEAR_ACC, line_width = 3)
    polyline.new(hist_bear_dec, line_color = COL_BEAR_DEC, line_width = 3)
    
    if hilbertEnable
        polyline.new(sine_pts, line_color = SINE_COLOR, line_width = 1)
        polyline.new(lead_pts, line_color = LEAD_COLOR, line_width = 1, line_style = line.style_dashed)

    // 5. MAIN CHART WEBBING (BUDGET PROTECTED - No Extrapolations)
    int dtw_lines = dtwEnable ? (dtwProject ? 3 * (1 + extrapInput) : 3) : 0
    int available_web_lines = 490 - dtw_lines
    int lines_per_step = 4 + (webBandsInput * 2) 
    if hpEnable
        lines_per_step += 1
    if sgEnable
        lines_per_step += 1
        
    int max_web_steps = math.max(1, available_web_lines / lines_per_step)
    
    int web_lookback = 120 
    int step = math.max(1, int(math.ceil(web_lookback * 1.0 / max_web_steps)))
    int start_i = math.max(step, math.min(lengthInput - 2, web_lookback))
    
    for i = start_i to step by step
        int idx1 = bar_index - i, int idx2 = bar_index - (i - step)
        
        float v_eU1 = zExtUp.get(i), float v_eU2 = zExtUp.get(i - step)
        float v_eD1 = zExtDn.get(i), float v_eD2 = zExtDn.get(i - step)
        float v_nw1 = zNW.get(i),    float v_nw2 = zNW.get(i - step)
        
        float v_mid1 = (v_eU1 + zIntUp.get(i) + zIntDn.get(i) + v_eD1) / 4.0
        float v_mid2 = (v_eU2 + zIntUp.get(i - step) + zIntDn.get(i - step) + v_eD2) / 4.0
        
        float vol1 = arr_volMods.get(i), float vol2 = arr_volMods.get(i - step)
        color nw_color_render = color.new(NW_COLOR, 0)
        
        line.new(idx1, v_eU1, idx2, v_eU2, color=color.new(BEAR_COLOR, 0), force_overlay=true, width=2)
        line.new(idx1, v_eD1, idx2, v_eD2, color=color.new(BULL_COLOR, 0), force_overlay=true, width=2)
        line.new(idx1, v_mid1, idx2, v_mid2, color=color.new(MID_COLOR, 60), force_overlay=true, style=line.style_dashed)
        line.new(idx1, v_nw1, idx2, v_nw2, color=nw_color_render, force_overlay=true, width=3)
        
        if hpEnable
            line.new(idx1, zHP.get(i), idx2, zHP.get(i - step), color=HP_COLOR, force_overlay=true, width=2)
        if sgEnable
            line.new(idx1, zSG.get(i), idx2, zSG.get(i - step), color=SG_COLOR, force_overlay=true, width=2)
        
        for b = 1 to webBandsInput
            float baseFrac = b / (webBandsInput + 1.0)
            float f1 = baseFrac * vol1, float f2 = baseFrac * vol2
            
            float w_up1 = v_mid1 + f1 * (v_eU1 - v_mid1), float w_dn1 = v_mid1 + f1 * (v_eD1 - v_mid1)
            float w_up2 = v_mid2 + f2 * (v_eU2 - v_mid2), float w_dn2 = v_mid2 + f2 * (v_eD2 - v_mid2)
            int transp = math.round(100.0 - (20.0 + baseFrac * 25.0))
            
            line.new(idx1, w_up1, idx2, w_up2, color=color.new(BEAR_COLOR, transp), force_overlay=true, width=1)
            line.new(idx1, w_dn1, idx2, w_dn2, color=color.new(BULL_COLOR, transp), force_overlay=true, width=1)

    // 6. HMM REGIME BACKGROUND FILL (Main Chart - No Extrapolations)
    if hmmEnable and hmmBgEnable
        float top_y = global_max_h + global_chart_h * 5.0
        float bot_y = math.max(0.0000001, global_min_l - global_chart_h * 5.0)
        
        int   start_regime_idx = na
        int   current_reg_state = na
        
        for i = lengthInput - 1 to 0
            int idx = bar_index - i
            float reg_val = zRegime.get(i)
            int state = reg_val > 0.3 ? 1 : (reg_val < -0.3 ? -1 : 0)
            
            if na(current_reg_state)
                current_reg_state := state
                start_regime_idx := idx
            else if state != current_reg_state or i == 0
                color bg_col = current_reg_state == 1 ? color.new(BULL_COLOR, 92) : (current_reg_state == -1 ? color.new(BEAR_COLOR, 92) : color.new(MID_COLOR, 95))
                box.new(left=start_regime_idx, top=top_y, right=idx, bottom=bot_y, border_color=na, bgcolor=bg_col, force_overlay=true)
                current_reg_state := state
                start_regime_idx := idx

    // 7. DTW FRACTAL TWIN MATCHER (DRAWN LAST TO PREVENT GARBAGE COLLECTION)
    if dtwEnable and lengthInput > dtwTempLen * 2
        array<float> all_dists = array.new_float(lengthInput, 1000000000.0)
        
        for search_i = dtwTempLen to lengthInput - dtwTempLen - 1
            float dist = 0.0
            for t = 0 to dtwTempLen - 1
                float val_temp = zNW.get(t)
                float val_search = zNW.get(search_i + t)
                dist += (val_temp - val_search) * (val_temp - val_search)
            all_dists.set(search_i, dist)
                
        array<int> sorted_indices = array.sort_indices(all_dists)
        array<int> top_matches = array.new_int()
        
        for s_idx = 0 to sorted_indices.size() - 1
            int candidate = sorted_indices.get(s_idx)
            if all_dists.get(candidate) >= 1000000000.0
                break
                
            bool distinct = true
            if top_matches.size() > 0
                for tm = 0 to top_matches.size() - 1
                    if math.abs(candidate - top_matches.get(tm)) < dtwTempLen
                        distinct := false
                        break
            
            if distinct
                top_matches.push(candidate)
                
            if top_matches.size() >= 3
                break
                
        for m_i = 0 to top_matches.size() - 1
            int best_match_idx = top_matches.get(m_i)
            int curr_idx = bar_index
            int twin_idx = bar_index - best_match_idx
            float curr_y = zNW.get(0)
            float twin_y = zNW.get(best_match_idx)
            
            int transp = m_i == 0 ? 0 : (m_i == 1 ? 40 : 70)
            int w      = m_i == 0 ? 3 : (m_i == 1 ? 2 : 1)
            color path_col = color.new(DTW_COLOR, transp)
            
            line.new(curr_idx, curr_y, twin_idx, twin_y, color=path_col, width=w, style=line.style_dashed, force_overlay=true)
            
            if m_i == 0
                label.new(twin_idx, twin_y, text="Primary Twin", style=label.style_label_down, color=color.new(color.black, 30), textcolor=path_col, size=size.small, tooltip="Dynamic Time Warping\nHistorical Structural Match", force_overlay=true)

            if dtwProject
                int p_len = math.min(extrapInput, best_match_idx)
                if p_len > 0
                    float prev_proj_val = curr_y
                    for k = 1 to p_len
                        float hist_future_val = zNW.get(best_match_idx - k)
                        float delta = hist_future_val - twin_y
                        float proj_val = curr_y + delta
                        line.new(curr_idx + k - 1, prev_proj_val, curr_idx + k, proj_val, color=path_col, width=w, style=line.style_dotted, force_overlay=true)
                        prev_proj_val := proj_val
                    if m_i == 0
                        label.new(curr_idx + p_len, prev_proj_val, text="Projected Path", style=label.style_label_up, color=color.new(color.black, 30), textcolor=path_col, size=size.tiny, tooltip="DTW Multiverse\nProjected Future Trajectory", force_overlay=true)

    // 8. HMM REGIME RIBBON (Subpane Base)
    if hmmEnable
        int hmm_step = math.max(1, int(math.ceil(lengthInput / 150.0))) 
        int hmm_start = math.max(hmm_step, math.min(lengthInput - 2, lengthInput))
        for i = hmm_start to hmm_step by hmm_step
            int idx1 = bar_index - i, int idx2 = bar_index - (i - hmm_step)
            float reg1 = zRegime.get(i)
            color col1 = reg1 > 0.3 ? BULL_COLOR : reg1 < -0.3 ? BEAR_COLOR : MID_COLOR
            line.new(idx1, -0.85, idx2, -0.85, color=col1, width=6, force_overlay=false)

    if showHUD
        table.cell(hud, 0, 0, "Log",      text_color = color.white, bgcolor = color.new(color.gray, 60), text_size=size.small)
        table.cell(hud, 1, 0, "Time",     text_color = color.white, bgcolor = color.new(color.gray, 60), text_size=size.small)
        table.cell(hud, 2, 0, "Signal",   text_color = color.white, bgcolor = color.new(color.gray, 60), text_size=size.small)
        table.cell(hud, 3, 0, "Z-Score",  text_color = color.white, bgcolor = color.new(color.gray, 60), text_size=size.small)
        table.cell(hud, 4, 0, "Conf.",    text_color = color.white, bgcolor = color.new(color.gray, 60), text_size=size.small)
        
        int total_sigs = log_bars.size()

        for row = 1 to 5

                int s_idx = sorted_idx.get(total_sigs - row)
